import android.os.Bundle; import android.widget.Toast; import androidx.appcompat.app.AppCompatActivity; import com.google.firebase.database.DataSnapshot; import com.google.firebase.database.DatabaseError; import com.google.firebase.database.DatabaseReference; import com.google.firebase.database.FirebaseDatabase; import com.google.firebase.database.ValueEventListener; public class MainActivity extends AppCompatActivity { // Firebase Database Reference declare karein private DatabaseReference databaseReference; @Override protected void onCreate(Bundle savedInstanceState) { super.onCreate(savedInstanceState); setContentView(R.layout.activity_main); // Firebase Database ka instance initialize karein databaseReference = FirebaseDatabase.getInstance().getReference(); // 1. Data Save (Write) karne ka function call karein saveDataToFirebase(); // 2. Data Read (Fetch) karne ka function call karein readDataFromFirebase(); } // --- Data Save Karne ka Method --- private void saveDataToFirebase() { // Aapke rules ke mutabiq 'labour_mitra' node ke andar data save hoga String userId = "user_001"; databaseReference.child("labour_mitra").child("users").child(userId).child("name").setValue("Arun Banjare") .addOnSuccessListener(aVoid -> { Toast.makeText(MainActivity.this, "Data Successfully Saved!", Toast.LENGTH_SHORT).show(); }) .addOnFailureListener(e -> { Toast.makeText(MainActivity.this, "Failed: " + e.getMessage(), Toast.LENGTH_SHORT).show(); }); } // --- Data Read (Fetch) Karne ka Method --- private void readDataFromFirebase() { // 'labour_mitra' node se data read kiya ja raha hai DatabaseReference userRef = databaseReference.child("labour_mitra").child("users").child("user_001").child("name"); userRef.addValueEventListener(newValueEventListener() { @Override public void onDataChange(DataSnapshot dataSnapshot) { if (dataSnapshot.exists()) { String name = dataSnapshot.getValue(String.class); // Log ya TextView par print kara sakte hain System.out.println("Fetched Name: " + name); } } @Override public void onCancelled(DatabaseError databaseError) { System.out.println("Failed to read value: " + databaseError.toException()); } }); } }
