package com.manpowerpro.erp; import jakarta.persistence.*; import lombok.*; import org.springframework.boot.SpringApplication; import org.springframework.boot.autoconfigure.SpringBootApplication; import org.springframework.data.jpa.repository.JpaRepository; import org.springframework.data.jpa.repository.Query; import org.springframework.data.repository.query.Param; import org.springframework.http.*; import org.springframework.stereotype.Repository; import org.springframework.stereotype.Service; import org.springframework.transaction.annotation.Transactional; import org.springframework.web.bind.annotation.*; import java.math.BigDecimal; import java.math.RoundingMode; import java.time.LocalDate; import java.time.LocalDateTime; import java.time.YearMonth; import java.util.*; import java.util.stream.Collectors; @SpringBootApplication public class LabourMitraApplication { public static void main(String[] args) { SpringApplication.run(LabourMitraApplication.class, args); } } // ============================================================================ // 1. ENUMS // ============================================================================ enum ShiftStatus { P, HD, A, PAID_LEAVE, UNPAID_LEAVE } enum WorkflowStatus { DRAFT, SUBMITTED, APPROVED, LOCKED } enum EmploymentStatus { ACTIVE, ON_LEAVE, SUSPENDED, TERMINATED } enum AccountStatus { ACTIVE, SUSPENDED, BLOCKED } enum RequestStatus { PENDING, APPROVED, REJECTED } // ============================================================================ // 2. REPOSITORIES // ============================================================================ @Repository interface WorkerRepository extends JpaRepository { List findByContractorIdAndStatus(Long contractorId, EmploymentStatus status); } @Repository interface AttendanceRecordRepository extends JpaRepository { @Query("SELECT a FROM AttendanceRecord a WHERE a.worker.contractor.id = :cId AND a.attendanceDate BETWEEN :sDate AND :eDate") List findMonthlyAttendance(@Param("cId") Long cId, @Param("sDate") LocalDate sDate, @Param("eDate") LocalDate eDate); } @Repository interface AdvanceKhurakiRepository extends JpaRepository { List findByWorkerContractorIdAndWageMonth(Long contractorId, String wageMonth); } @Repository interface PenaltyDeductionRepository extends JpaRepository { List findByWorkerIdAndWageMonth(Long workerId, String wageMonth); } // ============================================================================ // 4. DTO (PAYROLL SUMMARY) // ============================================================================ @Getter @Setter @NoArgsConstructor @AllArgsConstructor @Builder class WorkerPayrollSummaryDTO { private String workerId; private String fullName; private String uanNumber; private String tradeCategory; private BigDecimal dailyWageRate; private String bankAccount; private String ifscCode; private double presentShifts; private double totalOtHours; private BigDecimal grossWages; private BigDecimal employeePf12Cut; private BigDecimal statutoryNetPayable; // Sheet 2 (Gross - PF) private BigDecimal advanceKhurakiCut; private BigDecimal safetyPenaltyCut; private BigDecimal finalInHandPayout; // Sheet 3 (Gross - PF - Khuraki - Fine) } // ============================================================================ // 5. SERVICE LOGIC (SHEETS, NEFT, ECR) // ============================================================================ @Service @RequiredArgsConstructor class ManpowerPayrollService { private final WorkerRepository workerRepo; private final AttendanceRecordRepository attRepo; private final AdvanceKhurakiRepository advRepo; private final PenaltyDeductionRepository fineRepo; @Transactional(readOnly = true) public List computeMonthlySummary(Long contractorId, String wageMonthStr) { YearMonth ym = YearMonth.parse(wageMonthStr); List workers = workerRepo.findByContractorIdAndStatus(contractorId, EmploymentStatus.ACTIVE); List attendance = attRepo.findMonthlyAttendance(contractorId, ym.atDay(1), ym.atEndOfMonth()); List advances = advRepo.findByWorkerContractorIdAndWageMonth(contractorId, wageMonthStr); Map> attMap = attendance.stream().collect(Collectors.groupingBy(a -> a.getWorker().getId())); Map advMap = advances.stream().collect(Collectors.groupingBy(a -> a.getWorker().getId(), Collectors.reducing(BigDecimal.ZERO, AdvanceKhuraki::getAmount, BigDecimal::add))); List list = new ArrayList<>(); for (Worker w : workers) { List wAtt = attMap.getOrDefault(w.getId(), Collections.emptyList()); double shifts = 0.0, ot = 0.0; for (AttendanceRecord a : wAtt) { if (a.getStatus() == ShiftStatus.P) shifts += 1.0; else if (a.getStatus() == ShiftStatus.HD) shifts += 0.5; if (a.getOtHours() != null) ot += a.getOtHours(); } BigDecimal shiftEarned = w.getDailyWage().multiply(BigDecimal.valueOf(shifts)).setScale(2, RoundingMode.HALF_UP); BigDecimal otEarned = w.getDailyWage().divide(BigDecimal.valueOf(8), 4, RoundingMode.HALF_UP).multiply(BigDecimal.valueOf(ot)).setScale(2, RoundingMode.HALF_UP); BigDecimal gross = shiftEarned.add(otEarned); BigDecimal eePf = w.isPfApplicable() ? shiftEarned.multiply(BigDecimal.valueOf(0.12)).setScale(0, RoundingMode.HALF_UP) : BigDecimal.ZERO; BigDecimal statutoryNet = gross.subtract(eePf).max(BigDecimal.ZERO); BigDecimal advCut = advMap.getOrDefault(w.getId(), BigDecimal.ZERO); BigDecimal fineCut = fineRepo.findByWorkerIdAndWageMonth(w.getId(), wageMonthStr).stream().map(PenaltyDeduction::getAmount).reduce(BigDecimal.ZERO, BigDecimal::add); BigDecimal finalInHand = gross.subtract(eePf).subtract(advCut).subtract(fineCut).max(BigDecimal.ZERO); list.add(WorkerPayrollSummaryDTO.builder() .workerId(w.getWorkerId()).fullName(w.getFullName()).uanNumber(w.getUanNumber()) .tradeCategory(w.getTradeCategory()).dailyWageRate(w.getDailyWage()) .bankAccount(w.getBankAccount()).ifscCode(w.getIfscCode()) .presentShifts(shifts).totalOtHours(ot).grossWages(gross) .employeePf12Cut(eePf).statutoryNetPayable(statutoryNet) .advanceKhurakiCut(advCut).safetyPenaltyCut(fineCut).finalInHandPayout(finalInHand) .build()); } return list; } public String generateBankNeftCsv(Long contractorId, String wageMonth) { List list = computeMonthlySummary(contractorId, wageMonth); StringBuilder csv = new StringBuilder("Beneficiary_Account_No,IFSC_Code,Beneficiary_Name,Amount\n"); for (WorkerPayrollSummaryDTO d : list) { csv.append(String.format("\"%s\",\"%s\",\"%s\",%s\n", d.getBankAccount(), d.getIfscCode(), d.getFullName(), d.getFinalInHandPayout().setScale(0, RoundingMode.HALF_UP))); } return csv.toString(); } public String generateEpfoEcrText(Long contractorId, String wageMonth) { List list = computeMonthlySummary(contractorId, wageMonth); StringBuilder ecr = new StringBuilder(); for (WorkerPayrollSummaryDTO d : list) { if (d.getUanNumber() != null && !d.getUanNumber().isEmpty()) { int gross = d.getGrossWages().intValue(); int epfWage = d.getGrossWages().intValue(); int eePf = d.getEmployeePf12Cut().intValue(); int eps = (int) Math.round(epfWage * 0.0833); ecr.append(String.format("%s#~#%s#~#%d#~#%d#~#%d#~#%d#~#%d#~#%d#~#0\n", d.getUanNumber(), d.getFullName(), gross, epfWage, eePf, eps, eePf - eps, Math.max(0, 26 - (int) d.getPresentShifts()))); } } return ecr.toString(); } } // ============================================================================ // 6. REST API CONTROLLER // ============================================================================ @RestController @RequestMapping("/api/v1/contractor/payroll") @RequiredArgsConstructor @CrossOrigin(origins = "*") class ContractorPayrollController { private final ManpowerPayrollService payrollService; @GetMapping("/summary") public ResponseEntity> getSummary(@RequestParam Long contractorId, @RequestParam String wageMonth) { return ResponseEntity.ok(payrollService.computeMonthlySummary(contractorId, wageMonth)); } @GetMapping("/download/bank-neft") public ResponseEntity downloadNeft(@RequestParam Long contractorId, @RequestParam String wageMonth) { return ResponseEntity.ok() .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=Bulk_NEFT_" + wageMonth + ".csv") .contentType(MediaType.parseMediaType("text/csv")) .body(payrollService.generateBankNeftCsv(contractorId, wageMonth).getBytes()); } @GetMapping("/download/epfo-ecr") public ResponseEntity downloadEcr(@RequestParam Long contractorId, @RequestParam String wageMonth) { return ResponseEntity.ok() .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=EPFO_ECR_" + wageMonth + ".txt") .contentType(MediaType.TEXT_PLAIN) .body(payrollService.generateEpfoEcrText(contractorId, wageMonth).getBytes()); } }
