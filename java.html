package com.example.labourmitra; // Aapka package name import android.os.Bundle; import android.widget.Toast; import androidx.appcompat.app.AppCompatActivity; import com.google.firebase.database.DataSnapshot; import com.google.firebase.database.DatabaseError; import com.google.firebase.database.DatabaseReference; import com.google.firebase.database.FirebaseDatabase; import com.google.firebase.database.ValueEventListener; public class MainActivity extends AppCompatActivity { // Firebase Database Reference declare kar rahe hain private DatabaseReference databaseReference; @Override protected void onCreate(Bundle savedInstanceState) { super.onCreate(savedInstanceState); setContentView(R.layout.activity_main); // Firebase Database ka instance initialize kar rahe hain databaseReference = FirebaseDatabase.getInstance().getReference(); // 1. Database mein data save karne ke liye function call kiya saveDataToFirebase(); // 2. Database se data read (fetch) karne ke liye function call kiya readDataFromFirebase(); } // --- Data Save (Write) Karne ka Method --- private void saveDataToFirebase() { // 'labour_mitra' -> 'main_database' node ke andar user ka data save hoga String userId = "user_001"; databaseReference.child("labour_mitra").child("main_database").child("users").child(userId).child("name").setValue("Arun Banjare") .addOnSuccessListener(aVoid -> { // Jab data successfully save ho jayega Toast.makeText(MainActivity.this, "Data Successfully Saved to Main Database!", Toast.LENGTH_SHORT).show(); }) .addOnFailureListener(e -> { // Agar save karte waqt koi error aaya Toast.makeText(MainActivity.this, "Failed: " + e.getMessage(), Toast.LENGTH_SHORT).show(); }); } // --- Data Read (Fetch) Karne ka Method --- private void readDataFromFirebase() { // 'labour_mitra' -> 'main_database' node ke andar se specific user ka naam read kiya ja raha hai DatabaseReference userRef = databaseReference.child("labour_mitra").child("main_database").child("users").child("user_001").child("name"); userRef.addValueEventListener(new ValueEventListener() { @Override public void onDataChange(DataSnapshot dataSnapshot) { // Check kar rahe hain ki data wahan exist karta hai ya nahi if (dataSnapshot.exists()) { String name = dataSnapshot.getValue(String.class); // Console ya log mein print kara kar dekh sakte hain System.out.println("Fetched Name from Main Database: " + name); } } @Override public void onCancelled(DatabaseError databaseError) { // Agar read karte waqt koi error aaye System.out.println("Failed to read value: " + databaseError.toException()); } }); } }
