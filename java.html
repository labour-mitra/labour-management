package com.example.labourmitra; import android.os.Bundle; import android.widget.Toast; import androidx.appcompat.app.AppCompatActivity; import com.google.firebase.database.DataSnapshot; import com.google.firebase.database.DatabaseError; import com.google.firebase.database.DatabaseReference; import com.google.firebase.database.FirebaseDatabase; import com.google.firebase.database.ValueEventListener; import java.util.HashMap; import java.util.Map; public class MainActivity extends AppCompatActivity { // Firebase Database Root Reference (labour_mitra/main_database) private DatabaseReference mainDbRef; @Override protected void onCreate(Bundle savedInstanceState) { super.onCreate(savedInstanceState); setContentView(R.layout.activity_main); // Firebase Realtime Database path initialize kar rahe hain mainDbRef = FirebaseDatabase.getInstance().getReference("labour_mitra/main_database"); // Example: Correction requests ko contractor ke liye realtime sunna // listenToCorrectionRequests("CONT-001"); } // ========================================================================= // 1. SUPERVISOR PASSWORD RESET / UPDATE METHOD // ========================================================================= /** * Jab Contractor kisi Supervisor ka password change/reset karega: * Yeh method database me 'supervisors' node ke andar us Supervisor ka password update karega. */ public void resetSupervisorPassword(String supervisorId, String newPassword) { mainDbRef.child("supervisors").addListenerForSingleValueEvent(new ValueEventListener() { @Override public void onDataChange(DataSnapshot dataSnapshot) { if (dataSnapshot.exists()) { boolean found = false; for (DataSnapshot snapshot : dataSnapshot.getChildren()) { String id = snapshot.child("id").getValue(String.class); if (id != null && id.equals(supervisorId)) { // Password update kar rahe hain snapshot.getRef().child("password").setValue(newPassword) .addOnSuccessListener(aVoid -> { Toast.makeText(MainActivity.this, "सुपरवाइज़र पासवर्ड सफलतापूर्वक अपडेट हो गया!", Toast.LENGTH_SHORT).show(); }) .addOnFailureListener(e -> { Toast.makeText(MainActivity.this, "पासवर्ड अपडेट विफल: " + e.getMessage(), Toast.LENGTH_SHORT).show(); }); found = true; break; } } if (!found) { Toast.makeText(MainActivity.this, "Supervisor ID नहीं मिली!", Toast.LENGTH_SHORT).show(); } } } @Override public void onCancelled(DatabaseError databaseError) { Toast.makeText(MainActivity.this, "Error: " + databaseError.getMessage(), Toast.LENGTH_SHORT).show(); } }); } // ========================================================================= // 2. SUPERVISOR LOGIN VERIFICATION METHOD (NEW PASSWORD CHECK) // ========================================================================= /** * Supervisor Login ke waqt database password verify karne ka method: */ public void verifySupervisorLogin(String mobile, String enteredPassword, SupervisorLoginCallback callback) { mainDbRef.child("supervisors").addListenerForSingleValueEvent(new ValueEventListener() { @Override public void onDataChange(DataSnapshot dataSnapshot) { if (dataSnapshot.exists()) { boolean matched = false; for (DataSnapshot snapshot : dataSnapshot.getChildren()) { String dbMobile = snapshot.child("mobile").getValue(String.class); String dbPass = snapshot.child("password").getValue(String.class); if (dbMobile != null && dbMobile.equals(mobile)) { if (dbPass != null && dbPass.equals(enteredPassword)) { matched = true; String supName = snapshot.child("name").getValue(String.class); String contId = snapshot.child("contractorId").getValue(String.class); callback.onSuccess(supName, contId); } break; } } if (!matched) { callback.onFailure("अमान्य मोबाइल नंबर या पासवर्ड!"); } } else { callback.onFailure("डेटाबेस में कोई सुपरवाइज़र रिकॉर्ड नहीं मिला!"); } } @Override public void onCancelled(DatabaseError databaseError) { callback.onFailure("Database Error: " + databaseError.getMessage()); } }); } // ========================================================================= // 3. ATTENDANCE CORRECTION REQUEST BHEJNE KA METHOD // ========================================================================= /** * Jab Supervisor kisi worker ki locked attendance me correction request bhejega: */ public void sendAttendanceCorrectionRequest(String supervisorId, String contractorId, String workerId, String workerName, String date, String oldStatus, int oldOt, String newStatus, int newOt, String reason) { String reqId = "REQ-" + System.currentTimeMillis(); Map reqData = new HashMap<>(); reqData.put("id", reqId); reqData.put("supervisorId", supervisorId); reqData.put("contractorId", contractorId); reqData.put("workerId", workerId); reqData.put("workerName", workerName); reqData.put("date", date); reqData.put("oldStatus", oldStatus); reqData.put("oldOt", oldOt); reqData.put("newStatus", newStatus); reqData.put("newOt", newOt); reqData.put("reason", reason); reqData.put("status", "PENDING"); mainDbRef.child("correctionRequests").child(reqId).setValue(reqData) .addOnSuccessListener(aVoid -> { Toast.makeText(MainActivity.this, "सुधार निवेदन ठेकेदार को भेजा गया!", Toast.LENGTH_SHORT).show(); }) .addOnFailureListener(e -> { Toast.makeText(MainActivity.this, "निवेदन विफल: " + e.getMessage(), Toast.LENGTH_SHORT).show(); }); } // ========================================================================= // 4. CONTRACTOR REALTIME CORRECTION REQUEST LISTENER // ========================================================================= /** * Contractor side par pending requests ko realtime listen karne ke liye: */ public void listenToCorrectionRequests(String contractorId) { mainDbRef.child("correctionRequests").addValueEventListener(new ValueEventListener() { @Override public void onDataChange(DataSnapshot dataSnapshot) { if (dataSnapshot.exists()) { int pendingCount = 0; for (DataSnapshot snapshot : dataSnapshot.getChildren()) { String cId = snapshot.child("contractorId").getValue(String.class); String status = snapshot.child("status").getValue(String.class); if (contractorId.equals(cId) && "PENDING".equals(status)) { pendingCount++; } } // pendingCount ko aap apne Notification badge ya UI me set kar sakte hain } } @Override public void onCancelled(DatabaseError databaseError) { // Handle read error } }); } // Callback Interface for Login public interface SupervisorLoginCallback { void onSuccess(String supervisorName, String contractorId); void onFailure(String errorMessage); } }
